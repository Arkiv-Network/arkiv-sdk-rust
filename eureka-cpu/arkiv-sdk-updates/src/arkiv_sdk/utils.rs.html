15async fn transfer_funds(
16    provider: &DynProvider,
17    from: Address,
18    to: Address,
19    amount: U256,
20    retries: usize,
21) -> alloy::rpc::types::TransactionReceipt {
22    let tx = provider
23        .transaction_request()
24        .with_chain_id(provider.get_chain_id().await.unwrap())
25        .from(from)
26        .max_priority_fee_per_gas(1_000_000_000) // 1 gwei
27        .max_fee_per_gas(5_000_000_000) // 5 gwei
28        .gas_limit(2_800_000)
29        .to(to)
30        .value(amount);
31
32    let mut res: Result<TransactionReceipt, PendingTransactionError> =
33        Err(PendingTransactionError::FailedToRegister);
34    for attempt in 0..=retries {
35        let pending_tx = provider
36            .send_transaction(tx.clone())
37            .await
38            .expect("failed to send transaction");
39        match pending_tx.get_receipt().await {
40            Ok(receipt) => {
41                res = Ok(receipt);
42                break;
43            }
44            Err(err) if attempt == retries => {
45                res = Err(err);
46                break;
47            }
48            Err(_) => std::thread::sleep(std::time::Duration::from_secs(1)),
49        }
50    }
51    res.expect("failed to get receipt")
52}

54/// Only useful for local development and testing. Funds an account with some eth.
55/// The provider must not have a wallet linked to it, otherwise the provider will
56/// attempt to look for the signing credential of account index 0.
57pub async fn fund_account(
58    node_provider: &DynProvider,
59    address: Address,
60    amount: U256,
61) -> alloy::rpc::types::TransactionReceipt {
62    let from = node_provider
63        .get_accounts()
64        .await
65        .expect("failed to get accounts")
66        .first()
67        .expect("no accounts available on node")
68        .to_owned();
69    transfer_funds(node_provider, from, address, amount, 5).await
70}

72/// Generate the minimum required fee history for transactions to succeed. Use this when connecting to a dev node that has
73/// no history to pull gas information from. This can be useful if you are running into `receipts not found` errors.
74pub async fn generate_fee_history(node_provider: &DynProvider, http_endpoint: reqwest::Url) {
75    const MIN_FEE_HISTORY: std::ops::Range<i32> = 0..5;
76    let chain_id = node_provider
77        .get_chain_id()
78        .await
79        .expect("failed to get chain id from node provider");
80    let signer_pair = || async move {
81        let random_signer = || PrivateKeySigner::random().with_chain_id(Some(chain_id));
82        let alice = random_signer();
83        let bob = random_signer();
84        fund_account(
85            node_provider,
86            alice.address(),
87            U256::from_limbs([0, 100, 0, 0]),
88        )
89        .await;
90        (alice, bob)
91    };
92    for _ in MIN_FEE_HISTORY {
93        let (alice, bob) = signer_pair().await;
94        let wallet_provider = ProviderBuilder::new()
95            .with_chain_id(chain_id)
96            .wallet(alice.clone())
97            .connect_http(http_endpoint.clone())
98            .erased();
99
100        transfer_funds(
101            &wallet_provider,
102            alice.address(),
103            bob.address(),
104            U256::from_limbs([50, 0, 0, 0]),
105            5,
106        )
107        .await;
108    }
109}

111/// Converts an ETH amount to wei as a `U256`.
112/// Accepts a `BigDecimal` ETH value and returns the equivalent amount in wei as a `U256`.
113/// This is useful for preparing values for smart contract calls or transactions.
114/// Returns an error if the value is too large to fit in a `u128`.
115pub fn eth_to_wei(eth: BigDecimal) -> anyhow::Result<U256> {
116    let wei = (eth * BigDecimal::from(1_000_000_000_000_000_000u128))
117        .to_u128()
118        .ok_or_else(|| anyhow::anyhow!("Value too large"))?;
119    Ok(U256::from(wei))
120}

122/// Converts a wei amount (`U256`) to ETH as a `BigDecimal`.
123/// Useful for displaying human-readable ETH values from raw wei amounts, such as for UI or logs.
124/// Panics if the `U256` value cannot be parsed as a string (should not happen for valid values).
125pub fn wei_to_eth(wei: U256) -> BigDecimal {
126    BigDecimal::from_str(&wei.to_string()).unwrap()
127        / BigDecimal::from(1_000_000_000_000_000_000u128)
128}
