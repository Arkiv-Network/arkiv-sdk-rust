26/// Adds storage methods to a [`alloy::providers::Provider`].
27///
28/// [`alloy::network` Documentation](https://github.com/alloy-rs/alloy/tree/f341fd655a2ae230adcefee3f39a55e7b87ab620/crates/network)
29///
30/// This trait maintains dynamic compatibility with [`alloy::providers::DynProvider`].
31#[async_trait::async_trait]
32pub trait StorageProvider<S: StorageNetwork>: Provider<S> + Send + Sync {
33    /// Filter and retrieve entities matching a query expression.
34    ///
35    /// TODO:
36    /// When we make a query, the returned value from the request is still JSON,
37    /// however, this may be less idiomatic for practical use if the end user must
38    /// do non-trivial work to get a decompressed, decoded and deserialized payload.
39    fn query(
40        &self,
41        query_expression: &str,
42        options: QueryOpts,
43    ) -> RpcCall<serde_json::Value, serde_json::Value> {
44        self.client().request(
45            ArkivRpcMethod::Query,
46            serde_json::json!([query_expression, options]),
47        )
48    }
49
50    /// Returns total number of entities at current block.
51    fn get_entity_count(&self) -> RpcCall<(), u64> {
52        self.client().request(ArkivRpcMethod::GetEntityCount, ())
53    }
54
55    /// Returns storage slot usage accounting.
56    fn get_number_of_used_slots(&self) -> RpcCall<(), U256> {
57        self.client()
58            .request(ArkivRpcMethod::GetNumberOfUsedSlots, ())
59    }
60
61    /// Returns current block timing information.
62    fn get_block_timing(&self) -> RpcCall<(), BlockTiming> {
63        self.client().request(ArkivRpcMethod::GetBlockTiming, ())
64    }
65
66    /// Construct the default [`StorageNetwork::StorageTransactionRequest`].
67    fn storage_transaction(&self) -> S::StorageTransactionRequest {
68        Default::default()
69    }
70
71    async fn send_storage_transaction(
72        &self,
73        tx: S::StorageTransactionRequest,
74    ) -> TransportResult<PendingTransactionBuilder<S>> {
75        self.send_transaction(tx.into_request().with_chain_id(self.get_chain_id().await?))
76            .await
77    }
78
79    async fn create_entities(
80        &self,
81        creates: Vec<<<S::StorageTransactionRequest as StorageTransactionBuilder<S>>::Payload as PayloadBuilder<S>>::Create>,
82    ) -> TransportResult<PendingTransactionBuilder<S>> {
83        self.send_storage_transaction(self.storage_transaction().create_entities(creates))
84            .await
85    }
86
87    async fn update_entities(
88        &self,
89        updates: Vec<<<S::StorageTransactionRequest as StorageTransactionBuilder<S>>::Payload as PayloadBuilder<S>>::Update>,
90    ) -> TransportResult<PendingTransactionBuilder<S>> {
91        self.send_storage_transaction(self.storage_transaction().update_entities(updates))
92            .await
93    }
94
95    async fn delete_entities(
96        &self,
97        deletes: Vec<<<S::StorageTransactionRequest as StorageTransactionBuilder<S>>::Payload as PayloadBuilder<S>>::Delete>,
98    ) -> TransportResult<PendingTransactionBuilder<S>> {
99        self.send_storage_transaction(self.storage_transaction().delete_entities(deletes))
100            .await
101    }
102
103    async fn extend_entities(
104        &self,
105        extensions: Vec<<<S::StorageTransactionRequest as StorageTransactionBuilder<S>>::Payload as PayloadBuilder<S>>::Extend>,
106    ) -> TransportResult<PendingTransactionBuilder<S>> {
107        self.send_storage_transaction(self.storage_transaction().extend_entities(extensions))
108            .await
109    }
110
111    async fn transfer_entities(
112        &self,
113        transfers: Vec<<<S::StorageTransactionRequest as StorageTransactionBuilder<S>>::Payload as PayloadBuilder<S>>::Chown>,
114    ) -> TransportResult<PendingTransactionBuilder<S>> {
115        self.send_storage_transaction(self.storage_transaction().transfer_entities(transfers))
116            .await
117    }
118
119    /// A convenience method for subscribing to a stream of events from the [`StorageNetwork`]'s storage contract.
120    /// Provides a closure over a [`alloy::rpc::types::Filter`] with the [`crate::tx::StorageTransactionRequest::STORAGE_ADDRESS`] pre-populated
121    /// and attempts to convert the [`alloy::sol`] contract types into [`crate::network::StorageNetwork::StorageEvent`].
122    #[cfg(feature = "pubsub")]
123    async fn subscribe_storage_events(
124        &self,
125        f: fn(Filter) -> Filter,
126    ) -> TransportResult<
127        futures::stream::Map<
128            alloy::pubsub::SubscriptionStream<Log>,
129            fn(Log) -> Result<S::StorageEvent, <S::StorageEvent as TryFrom<Log>>::Error>,
130        >,
131    > {
132        let subscription = self
133            .subscribe_logs(&f(
134                Filter::new().address(S::StorageTransactionRequest::STORAGE_ADDRESS)
135            ))
136            .await?;
137        Ok(subscription.into_stream().map(S::StorageEvent::try_from))
138    }
139}
