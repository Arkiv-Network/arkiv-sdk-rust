3/// Available RPC methods for interacting with the Arkiv network.
4#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
5pub enum ArkivRpcMethod {
6    /// `arkiv_query`
7    ///
8    /// Filter and retrieve entities matching a query expression.
9    Query,
10    /// `arkiv_getEntityCount`
11    ///
12    /// Returns total number of entities at current block.
13    GetEntityCount,
14    /// `arkiv_getNumberOfUsedSlots`
15    ///
16    /// Returns storage slot usage accounting.
17    GetNumberOfUsedSlots,
18    /// `arkiv_getBlockTiming`
19    ///
20    /// Returns current block timing information.
21    GetBlockTiming,
22}
23impl ArkivRpcMethod {
24    const ARKIV_QUERY: &str = "arkiv_query";
25    const ARKIV_GET_ENTITY_COUNT: &str = "arkiv_getEntityCount";
26    const ARKIV_GET_NUMBER_OF_USED_SLOTS: &str = "arkiv_getNumberOfUsedSlots";
27    const ARKIV_GET_BLOCK_TIMING: &str = "arkiv_getBlockTiming";
28
29    /// Return the corresponding method as a `&'static str`.
30    pub const fn as_str(&self) -> &'static str {
31        match self {
32            Self::Query => Self::ARKIV_QUERY,
33            Self::GetEntityCount => Self::ARKIV_GET_ENTITY_COUNT,
34            Self::GetNumberOfUsedSlots => Self::ARKIV_GET_NUMBER_OF_USED_SLOTS,
35            Self::GetBlockTiming => Self::ARKIV_GET_BLOCK_TIMING,
36        }
37    }
38}
39impl std::ops::Deref for ArkivRpcMethod {
40    type Target = str;
41    fn deref(&self) -> &Self::Target {
42        self.as_str()
43    }
44}
45impl From<ArkivRpcMethod> for std::borrow::Cow<'static, str> {
46    fn from(value: ArkivRpcMethod) -> Self {
47        value.as_str().into()
48    }
