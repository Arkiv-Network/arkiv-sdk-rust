2//! Types used by the arkiv storage network for interacting with entities via RPC.
3
4use serde::{Deserialize, Serialize};
5
6mod event;
7#[cfg(feature = "pubsub")]
8pub use event::*;
9
10#[derive(Debug, Default, Clone, PartialEq, Eq, Serialize, Deserialize)]
11pub struct QueryOpts {
12    /// Historical block number for query
13    pub at_block: Option<u64>,
14    /// Which fields to return
15    pub include_data: Option<IncludeData>,
16    /// Ordering by attributes
17    pub order_by: Option<Vec<OrderByAttribute>>,
18    /// Pagination limit
19    pub results_per_page: Option<u64>,
20    /// Pagination cursor from previous response
21    pub cursor: Option<String>,
22}
23
24#[derive(Debug, Default, Clone, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize)]
25pub struct IncludeData {
26    /// Entity key hash
27    pub key: bool,
28    /// User-defined attributes
29    pub attributes: bool,
30    /// System-generated attributes ($sequence, $creator)
31    pub synthetic_attributes: bool,
32    /// Entity data content
33    pub payload: bool,
34    /// MIME type
35    pub content_type: bool,
36    /// Expiration block number
37    pub expiration: bool,
38    /// Owner address
39    pub owner: bool,
40    /// Creation block number
41    pub created_at_block: bool,
42    /// Last modification block
43    pub last_modified_at_block: bool,
44    /// Transaction index
45    pub transaction_index_in_block: bool,
46    /// Operation index
47    pub operation_index_in_transaction: bool,
48}
49
50#[derive(Debug, Default, Clone, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize)]
51pub struct OrderByAttribute {
52    pub name: String,
53    #[serde(rename = "type")]
54    pub r#type: String,
55    #[serde(rename = "desc")]
56    pub descending: bool,
57}
58
59#[derive(Debug, Default, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize)]
60#[serde(rename_all = "snake_case")]
61pub struct BlockTiming {
62    pub current_block: u64,
63    pub current_block_time: u64,
64    #[serde(rename = "duration")]
65    pub block_duration: u64,
66}
