6/// The Ethereum address of the Arkiv storage contract. All entity-related transactions are sent to this address.
7///
8/// This address does not need to be provided when sending a transaction. The [`crate::tx::StorageTransactionBuilder`]
9/// trait implementation for [`crate::tx::StorageTransactionRequest`] uses this address internally.
10///
11/// For convenience, this address can be referred to directly from the [`crate::tx::StorageTransactionRequest`] itself.
12pub const STORAGE_ADDRESS: alloy::primitives::Address =
13    alloy::primitives::address!("0x00000000000000000000000000000061726B6976");
14
15alloy::sol! {
16    /// The storage contract ABI for the Arkiv network. Contains bindings to event data used in decoding event logs and receipts.
17    /// The version of this contract is taken directly from this commit: <https://github.com/Arkiv-Network/arkiv-op-geth/blob/4d831af4042037469d9b941e41cd79d78fd37f4c/arkiv/logs/arkiv.go>.
18    contract ArkivAbi {
19        /// ArkivEntityCreated is the event signature for entity creation logs.
20        ///
21        /// Parameters: entityKey (indexed), ownerAddress(indexed), expirationBlock, cost (wei)
22        ///
23        event ArkivEntityCreated(
24            uint256 indexed entityKey,
25            address indexed ownerAddress,
26            uint256 expirationBlock,
27            uint256 cost,
28        );
29
30        /// ArkivEntityUpdated is the event signature for entity update logs.
31        ///
32        /// Parameters: entityKey (indexed), ownerAddress(indexed), oldExpirationBlock, newExpirationBlock, cost (wei)
33        ///
34        event ArkivEntityUpdated(
35            uint256 indexed entityKey,
36            address indexed ownerAddress,
37            uint256 oldExpirationBlock,
38            uint256 newExpirationBlock,
39            uint256 cost,
40        );
41
42        /// ArkivEntityExpired is the event signature for entity expiration logs.
43        ///
44        /// Parameters: entityKey (indexed), ownerAddress(indexed)
45        ///
46        event ArkivEntityExpired(
47            uint256 indexed entityKey,
48            address indexed ownerAddress,
49        );
50
51        /// ArkivEntityDeleted is the event signature for entity deletion logs.
52        ///
53        /// Parameters: entityKey (indexed), ownerAddress(indexed)
54        ///
55        event ArkivEntityDeleted(
56            uint256 indexed entityKey,
57            address indexed ownerAddress,
58        );
59
60        /// ArkivEntityBTLExtended is the event signature for extending BTL of an entity.
61        ///
62        /// Parameters: entityKey (indexed), ownerAddress(indexed), oldExpirationBlock, newExpirationBlock, cost (wei)
63        ///
64        event ArkivEntityBTLExtended(
65            uint256 indexed entityKey,
66            address indexed ownerAddress,
67            uint256 oldExpirationBlock,
68            uint256 newExpirationBlock,
69            uint256 cost,
70        );
71
72        /// ArkivEntityOwnerChanged is the event signature for changing the owner of an entity.
73        ///
74        /// Parameters: entityKey (indexed), oldOwnerAddress(indexed), newOwnerAddress(indexed)
75        ///
76        event ArkivEntityOwnerChanged(
77            uint256 indexed entityKey,
78            address indexed oldOwnerAddress,
79            address indexed newOwnerAddress,
80        );
81    }
